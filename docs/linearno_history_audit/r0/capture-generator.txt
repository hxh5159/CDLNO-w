def sha(t):
    a=t.detach().cpu().contiguous()
    return hashlib.sha256(a.numpy().tobytes()).hexdigest()

def describe(t):
    if t is None: return None
    a=t.detach().cpu()
    assert torch.isfinite(a).all()
    return dict(shape=list(a.shape),dtype=str(a.dtype),sha256=sha(a),
                norm=float(a.double().norm()),sum=float(a.double().sum()))

def rng():
    npstate=np.random.get_state()
    return dict(torch=torch.get_rng_state().tolist(),python=random.getstate(),
                numpy=[npstate[0],npstate[1].tolist(),*npstate[2:]])

def rhash():
    return hashlib.sha256(json.dumps(rng(),sort_keys=True).encode()).hexdigest()
